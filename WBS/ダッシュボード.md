```dataviewjs
const DAY = 86400000;
const DT = dv.luxon.DateTime;
const MEMBERS = 1;          // ★メンバー数。増減したらここを変える
const HOURS_PER_DAY = 7;    // ★1人1日の作業可能時間

const all   = dv.pages('"WBS/タスク"').where(p => p.due).array();
const isMS  = p => p.file.name.startsWith("【MS】");
const tasks = all.filter(p => !isMS(p));
const today = DT.local().startOf("day").toMillis();
const sumH  = arr => Math.round(arr.reduce((s, p) => s + (p.estimate ?? 0), 0) * 10) / 10;

// --- 遅れ：期日を過ぎているのに終わっていないタスクの見積合計 ---
const late  = tasks.filter(p => p.status !== "完了" && p.due.toMillis() < today);
const lateH = sumH(late);

// --- 全体の進み：見積ベースの消化率 ---
const totalH = sumH(tasks);
const doneH  = sumH(tasks.filter(p => p.status === "完了"));
const pctDone = totalH ? Math.round(doneH / totalH * 100) : 0;

// --- 次のマイルストーン ---
const bizDays = (fromMs, toMs) => {
  let n = 0;
  for (let d = DT.fromMillis(fromMs); d.toMillis() <= toMs; d = d.plus({ days: 1 }))
    if (d.weekday <= 5) n++;
  return n;
};
const nextMS = all
  .filter(p => isMS(p) && p.status !== "完了" && p.due.toMillis() >= today)
  .sort((a, b) => a.due.toMillis() - b.due.toMillis())[0];

const card = (title, big, sub, color) => `
  <div style="flex:1 1 200px;padding:12px 14px;border:1px solid var(--background-modifier-border);
              border-radius:8px;background:var(--background-secondary)">
    <div style="font-size:11px;color:var(--text-muted)">${title}</div>
    <div style="font-size:22px;font-weight:700;line-height:1.3;margin:2px 0;color:${color}">${big}</div>
    <div style="font-size:11px;color:var(--text-muted)">${sub}</div>
  </div>`;

// 遅れカード
const cLate = lateH === 0
  ? card("遅れ", "なし", "期日を過ぎた未完了タスクは0件", "var(--color-green,#3aa675)")
  : card("遅れ", `${lateH} 時間`, `${late.length}件が期日超過`, "var(--color-red,#e05252)");

// 進捗カード
const cProg = card("全体の進み（見積ベース）", `${pctDone}%`,
  `完了 ${doneH}h / 全体 ${totalH}h`, "var(--text-normal)");

// マイルストーンカード
let cMS;
if (!nextMS) {
  cMS = card("次のマイルストーン", "未設定",
    "タスク名の頭に【MS】を付けると出ます", "var(--text-muted)");
} else {
  const msMs   = nextMS.due.toMillis();
  const remain = bizDays(today, msMs);
  const cap    = remain * MEMBERS * HOURS_PER_DAY;
  const need   = sumH(tasks.filter(p => p.status !== "完了" && p.due.toMillis() <= msMs));
  const ratio  = cap ? need / cap : 99;
  const [verdict, col] =
      ratio <= 0.7 ? ["間に合う",     "var(--color-green,#3aa675)"]
    : ratio <= 1.0 ? ["ぎりぎり",     "var(--color-yellow,#c9a227)"]
    :                ["間に合わない", "var(--color-red,#e05252)"];
  cMS = card(`次のマイルストーン：${nextMS.file.name}`, verdict,
    `${nextMS.due.toFormat("M/d")}まで 残り${remain}営業日 ｜ 必要 ${need}h / 使える ${cap}h`, col);
}

dv.el("div", "").innerHTML =
  `<div style="display:flex;gap:10px;flex-wrap:wrap">${cLate}${cProg}${cMS}</div>`;
```

> 「使える時間」は **残り営業日 × 1人 × 7h** の単純計算です。会議や手戻りは含みません。
> だから **`ぎりぎり` が出た時点で実際は危ない**と思ってください。

## 📊 ガント

上のボタンで **「工程別」と「担当別」を切り替え**られます。
タスクを増やすと自動で伸びます。**再生成の操作は要りません。**

🟣 未着手 / 🔵 対応中 / ⚪ 完了。🔴 赤い棒＝**期日超過**、🟠 橙＝**値が不正**（どちらも自動判定）。
◆ がマイルストーン。**黄色い破線**がその期日を全行に通しています。赤い縦線が今日。

```dataviewjs
const DAY = 86400000;
const DT = dv.luxon.DateTime;
const CATS = ["1_プロジェクト管理", "2_AI管理", "3_要件定義", "4_基本設計",
              "5_詳細設計・実装・テスト", "6_結合テスト観点", "7_インフラ連携", "8_システムテスト＋検収",
              "9_改善"];
const STATUS = ["未着手", "対応中", "完了"];
const isMS = p => p.file.name.startsWith("【MS】");

const pages = dv.pages('"WBS/タスク"').where(p => p.due).array();

if (!pages.length) {
  dv.paragraph("タスクがまだありません。`/wbs-new` で作ってください。");
} else {
  const DAY_W = 26;                                     // ★1日の横幅(px)
  const startOf = p => (p.start ?? p.due).toMillis();
  const endOf   = p => p.due.toMillis() + DAY;          // 期日当日を含める

  const min  = Math.min(...pages.map(startOf)) - DAY;
  const max  = Math.max(...pages.map(endOf))   + DAY;
  const days = Math.round((max - min) / DAY);
  const trackW = days * DAY_W;
  const x = t => Math.round((t - min) / DAY) * DAY_W;
  const today = DT.local().startOf("day").toMillis();

  const colorOf = p =>
      !STATUS.includes(p.status)   ? "var(--color-orange, #d08a3e)"   // 状態の値が不正
    : p.status === "完了"          ? "var(--text-faint, #999)"
    : p.due.toMillis() < today     ? "var(--color-red, #e05252)"      // 期日超過（自動判定）
    : isMS(p)                      ? "var(--color-yellow, #c9a227)"   // マイルストーン
    : p.status === "対応中"        ? "var(--color-blue, #4a7fd4)"
    :                                "var(--color-purple, #8b80c9)";  // 未着手

  // 1日ごとの目盛り
  const WD = "月火水木金土日";
  const cells = [];
  for (let i = 0; i < days; i++) {
    const d = DT.fromMillis(min).plus({ days: i });
    cells.push({ i, d, weekend: d.weekday >= 6 });
  }

  const grid = cells.map(c =>
    `<div style="position:absolute;left:${c.i * DAY_W}px;width:${DAY_W}px;top:0;bottom:0;
      border-left:1px solid var(--background-modifier-border);
      ${c.weekend ? "background:var(--background-modifier-hover);" : ""}"></div>`).join("");

  // マイルストーンの縦破線（全行に通す）
  const msLines = pages.filter(isMS).map(p =>
    `<div style="position:absolute;left:${x(p.due.toMillis())}px;top:0;bottom:0;
      border-left:1px dashed var(--color-yellow,#c9a227);opacity:.85"></div>`).join("");

  const todayLine = (today >= min && today < max)
    ? `<div style="position:absolute;left:${x(today)}px;top:0;bottom:0;border-left:2px solid var(--color-red,#e05252);opacity:.7"></div>` : "";

  const header =
    cells.filter(c => c.i === 0 || c.d.day === 1).map(c =>
      `<div style="position:absolute;left:${c.i * DAY_W + 2}px;top:0;font-size:11px;font-weight:600;
        color:var(--text-muted);white-space:nowrap">${c.d.toFormat("M月")}</div>`).join("")
    + cells.map(c =>
      `<div style="position:absolute;left:${c.i * DAY_W}px;width:${DAY_W}px;top:14px;text-align:center;
        font-size:10px;line-height:1.25;
        color:${c.weekend ? "var(--text-faint,#999)" : "var(--text-muted)"}">${c.d.day}<br>${WD[c.d.weekday - 1]}</div>`).join("");

  // グループ分け。マイルストーンは常に先頭、その下を「工程順」か「担当順」で並べる
  const owners = [...new Set(pages.filter(p => !isMS(p)).map(p => p.owner ?? "(未割当)"))].sort();

  const groupOf = (p, mode) =>
      isMS(p)         ? "🚩 マイルストーン"
    : mode === "担当" ? (p.owner ?? "(未割当)")
    : (CATS.includes(p.category) ? p.category : `⚠️ 区分が不正: ${p.category ?? "(空)"}`);

  const rank = (p, mode) => {
    if (isMS(p)) return -1;
    const list = mode === "担当" ? owners : CATS;
    const key  = mode === "担当" ? (p.owner ?? "(未割当)") : p.category;
    const i = list.indexOf(key);
    return i < 0 ? 99 : i;
  };

  // 2列目：工程別のときは担当名、担当別のときは工程名を出す
  const subOf = (p, mode) => mode === "担当"
    ? String(p.category ?? "").replace(/^\d_/, "")
    : (p.owner ?? "");

  const build = (mode) => {

  const sorted = pages.slice().sort((a, b) =>
    rank(a, mode) - rank(b, mode) || startOf(a) - startOf(b) || endOf(a) - endOf(b));

  let lastGroup = null;
  const body = sorted.map(p => {
    let head = "";
    const g = groupOf(p, mode);
    if (g !== lastGroup) {
      lastGroup = g;
      head = `<div style="grid-column:1/-1;margin-top:6px;padding:2px 0;font-size:12px;font-weight:600;
                          color:var(--text-muted);border-bottom:1px solid var(--background-modifier-border)">${g}</div>`;
    }

    const mark = isMS(p)
      ? `<div title="${p.file.name}（${p.due.toFormat("M/d")}）"
              style="position:absolute;left:${x(p.due.toMillis()) - 6}px;top:5px;width:12px;height:12px;
                     transform:rotate(45deg);background:${colorOf(p)};
                     border:1px solid var(--background-modifier-border)"></div>`
      : (() => {
          const l = x(startOf(p));
          const w = Math.max(x(endOf(p)) - l, 6);
          return `<div title="${p.file.name}（${p.estimate ?? "?"}h）"
                       style="position:absolute;left:${l}px;width:${w}px;top:4px;height:14px;border-radius:3px;
                              background:${colorOf(p)};border:1px solid var(--background-modifier-border)"></div>`;
        })();

    return head + `
      <div style="font-size:12px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap;padding-left:8px;
                  ${isMS(p) ? "font-weight:600;" : ""}">
        <a class="internal-link" data-href="${p.file.path}" href="${p.file.path}">${p.file.name}</a>
      </div>
      <div style="font-size:11px;color:var(--text-muted);white-space:nowrap;overflow:hidden;text-overflow:ellipsis">${subOf(p, mode)}</div>
      <div style="position:relative;height:22px;width:${trackW}px">${grid}${msLines}${todayLine}${mark}</div>`;
  }).join("");

  return `
    <div style="overflow-x:auto">
      <div style="display:grid;grid-template-columns:minmax(160px,220px) 100px ${trackW}px;gap:2px 8px;align-items:center;width:max-content">
        <div></div><div></div><div style="position:relative;height:40px;width:${trackW}px">${header}</div>
        ${body}
      </div>
    </div>`;
  };

  // --- 「工程別 / 担当別」の切り替えボタン ---
  const root   = dv.el("div", "");
  const btnBar = document.createElement("div");
  const chart  = document.createElement("div");
  btnBar.style.cssText = "display:flex;gap:6px;margin-bottom:8px";
  root.appendChild(btnBar);
  root.appendChild(chart);

  let mode = "工程";
  const btns = {};
  const draw = () => {
    chart.innerHTML = build(mode);
    for (const m of Object.keys(btns)) {
      const on = m === mode;
      btns[m].style.cssText = `padding:4px 14px;font-size:12px;border-radius:6px;cursor:pointer;
        border:1px solid var(--background-modifier-border);
        background:${on ? "var(--interactive-accent)" : "var(--background-secondary)"};
        color:${on ? "var(--text-on-accent)" : "var(--text-muted)"};
        font-weight:${on ? "600" : "400"};`;
    }
  };
  for (const m of ["工程", "担当"]) {
    const b = document.createElement("button");
    b.textContent = `${m}別`;
    b.onclick = () => { mode = m; draw(); };
    btns[m] = b;
    btnBar.appendChild(b);
  }
  draw();
}
```

## 🗓 個人別 × 日別の時間

**誰のどの日が詰まっているか**を見る表です。上のボタンで `見積 / 実績` と表示期間を切り替えられます。

各タスクの時間を **`start`〜`due` の営業日（土日を除く）に均等割り**した概算です。
日ごとの入力はしていないので、**実際に何時にやったかではありません。**
🟡 は1日 5h超、🔴 は **7h超（＝入れすぎ）**。セルにマウスを乗せると内訳のタスク名が出ます。

```dataviewjs
const DT   = dv.luxon.DateTime;
const CAP  = 7;                 // ★1人1日の作業可能時間。ここを変えると赤の基準が変わる
const WARN = 5;                 // ★黄色の基準
const WD   = "月火水木金土日";
const isMS = p => p.file.name.startsWith("【MS】");
const r1   = v => Math.round(v * 10) / 10;

const pages = dv.pages('"WBS/タスク"').array().filter(p => p.due && !isMS(p));

if (!pages.length) {
  dv.paragraph("タスクがまだありません。`/wbs-new` で作ってください。");
} else {
  // --- 各タスクの見積/実績を、start〜due の営業日に均等割りする ---
  const cell = {};              // cell[担当][yyyy-MM-dd] = { est, act, items }
  for (const p of pages) {
    const owner = p.owner ?? "(未割当)";
    let s = p.start ?? p.due;
    const e = p.due;
    if (s.toMillis() > e.toMillis()) s = e;          // 開始 > 期日 の打ち間違いは期日だけに寄せる

    let days = [];
    for (let d = s; d.toMillis() <= e.toMillis(); d = d.plus({ days: 1 })) days.push(d);
    const biz = days.filter(d => d.weekday <= 5);
    if (biz.length) days = biz;                      // 土日しかない期間はそのまま土日に載せる

    const est = (p.estimate ?? 0) / days.length;
    const act = (p.actual   ?? 0) / days.length;
    for (const d of days) {
      const o = cell[owner] ?? (cell[owner] = {});
      const k = d.toFormat("yyyy-MM-dd");
      const c = o[k] ?? (o[k] = { est: 0, act: 0, items: [] });
      c.est += est; c.act += act;
      c.items.push(`${p.file.name}（見積${r1(est)}h/実績${r1(act)}h）`);
    }
  }

  const owners  = Object.keys(cell).sort();
  const allKeys = [...new Set(owners.flatMap(o => Object.keys(cell[o])))].sort();
  const minD  = DT.fromISO(allKeys[0]);
  const maxD  = DT.fromISO(allKeys[allKeys.length - 1]);
  const today = DT.local().startOf("day");

  const rangeOf = (span) => {
    if (span === "全期間") return [minD, maxD];
    const mon = today.startOf("week");               // luxonは月曜始まり
    return span === "今週" ? [mon, mon.plus({ days: 6 })] : [mon, mon.plus({ days: 27 })];
  };

  const bg = (v) =>
      v > CAP  ? "background:var(--color-red,#e05252);color:#fff;font-weight:600;"
    : v > WARN ? "background:var(--color-yellow,#c9a227);color:#000;"
    : "";

  const build = (kind, span) => {
    const [from, to] = rangeOf(span);
    const cols = [];
    for (let d = from; d.toMillis() <= to.toMillis(); d = d.plus({ days: 1 })) cols.push(d);

    const key = kind === "実績" ? "act" : "est";
    const get = (own, d) => cell[own]?.[d.toFormat("yyyy-MM-dd")];

    const th = (html, style = "") =>
      `<th style="border:1px solid var(--background-modifier-border);padding:2px 6px;
                  font-size:11px;font-weight:600;color:var(--text-muted);${style}">${html}</th>`;

    const head = th("担当",
        "position:sticky;left:0;z-index:2;background:var(--background-secondary);min-width:110px;text-align:left")
      + cols.map(d => th(
          `${d.toFormat("M/d")}<br>${WD[d.weekday - 1]}`,
          `min-width:38px;text-align:center;` +
          (d.weekday >= 6 ? "background:var(--background-modifier-hover);" : "") +
          (d.hasSame(today, "day") ? "border-bottom:2px solid var(--color-red,#e05252);" : "")
        )).join("")
      + th("合計", "background:var(--background-secondary);text-align:right;min-width:52px");

    const colSum = cols.map(() => 0);
    const rows = owners.map(own => {
      let sum = 0;
      const tds = cols.map((d, i) => {
        const c = get(own, d);
        const v = r1(c ? c[key] : 0);
        sum += v; colSum[i] += v;
        const tip = v && c ? ` title="${c.items.join(" ／ ").replace(/"/g, "'")}"` : "";
        return `<td${tip} style="border:1px solid var(--background-modifier-border);padding:2px 4px;
                  text-align:center;font-size:11px;
                  ${d.weekday >= 6 && !v ? "background:var(--background-modifier-hover);" : bg(v)}">${v || ""}</td>`;
      }).join("");
      return `<tr>
        <td style="position:sticky;left:0;z-index:1;background:var(--background-primary);
                   border:1px solid var(--background-modifier-border);padding:2px 6px;font-size:12px;
                   white-space:nowrap">${own}</td>
        ${tds}
        <td style="border:1px solid var(--background-modifier-border);padding:2px 6px;text-align:right;
                   font-size:12px;font-weight:600">${r1(sum)}</td></tr>`;
    }).join("");

    const total = r1(colSum.reduce((a, b) => a + b, 0));
    const foot = `<tr>
      <td style="position:sticky;left:0;z-index:1;background:var(--background-secondary);
                 border:1px solid var(--background-modifier-border);padding:2px 6px;font-size:11px;
                 color:var(--text-muted)">日計（${owners.length}人）</td>
      ${colSum.map((v, i) => `<td style="border:1px solid var(--background-modifier-border);padding:2px 4px;
          text-align:center;font-size:11px;color:var(--text-muted);
          ${cols[i].weekday >= 6 ? "background:var(--background-modifier-hover);" : ""}">${r1(v) || ""}</td>`).join("")}
      <td style="border:1px solid var(--background-modifier-border);padding:2px 6px;text-align:right;
                 font-size:12px;font-weight:700">${total}</td></tr>`;

    return `<div style="overflow-x:auto">
      <table style="border-collapse:collapse;width:max-content">
        <thead><tr>${head}</tr></thead><tbody>${rows}${foot}</tbody></table></div>`;
  };

  // --- 切り替えボタン（集計＝見積/実績、期間＝今週/4週間/全期間） ---
  const root  = dv.el("div", "");
  const bar   = document.createElement("div");
  const table = document.createElement("div");
  bar.style.cssText = "display:flex;gap:6px;flex-wrap:wrap;margin-bottom:8px";
  root.appendChild(bar); root.appendChild(table);

  let kind = "見積", span = "4週間";
  const btns = [];
  const draw = () => {
    table.innerHTML = build(kind, span);
    for (const b of btns) {
      const on = b.dataset.group === "kind" ? b.dataset.value === kind : b.dataset.value === span;
      b.style.cssText = `padding:4px 12px;font-size:12px;border-radius:6px;cursor:pointer;
        border:1px solid var(--background-modifier-border);
        background:${on ? "var(--interactive-accent)" : "var(--background-secondary)"};
        color:${on ? "var(--text-on-accent)" : "var(--text-muted)"};
        font-weight:${on ? "600" : "400"};`;
    }
  };
  const addBtns = (group, values) => {
    for (const v of values) {
      const b = document.createElement("button");
      b.textContent = v;
      b.dataset.group = group; b.dataset.value = v;
      b.onclick = () => { if (group === "kind") kind = v; else span = v; draw(); };
      btns.push(b); bar.appendChild(b);
    }
  };
  addBtns("kind", ["見積", "実績"]);
  const sep = document.createElement("div");
  sep.style.cssText = "width:1px;background:var(--background-modifier-border);margin:0 4px";
  bar.appendChild(sep);
  addBtns("span", ["今週", "4週間", "全期間"]);
  draw();
}
```

> **赤は「その日に7h超の予定を積んでいる」という意味です。** 期日を後ろに倒すか、担当を替えるか、
> タスクを分割するかのどれかを朝会で決めてください。
> 実績側は `actual` を期間で割った概算なので、**1日単位の実績としては使えません**（傾向を見るだけ）。
